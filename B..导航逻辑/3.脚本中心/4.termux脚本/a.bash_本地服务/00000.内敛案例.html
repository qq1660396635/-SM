#!/bin/bash
# 1. 一键中文分析报告：调用树+ 符号列表，结果与源文件同目录
set -euo pipefail

# 2. 读取用户输入的文件路径，并进行有效性检查
read -r -p "请输入要分析的C 源文件路径（支持空格）: " SRC
[[ -z "$SRC" ]] && { echo " 未输入路径，退出"; exit 1; }
SRC="${SRC%\"}"
SRC="${SRC#\"}"
[[ -f "$SRC" ]] || { echo " 文件不存在：$SRC"; exit 1; }

# 3. 获取文件信息并构建输出报告路径
DIR=$(dirname "$SRC")
BASE=$(basename "$SRC" | sed 's/\.[^.]*$//')
REPORT="$DIR/${BASE}_分析报告.md"

# 4. 写入Markdown 报告的头部信息
cat > "$REPORT" < 文件：\`$SRC\`
> 生成时间：$(date "+%Y-%m-%d %H:%M:%S")
---
## 一、函数调用关系
EOF

# 5. 生成函数调用树部分，并添加序号
echo "🔄生成调用树..."
if ! cflow "$SRC" 2>/dev/null | awk '
BEGIN{idx=1}
/^[^ \t]/{gsub(/<.*>/,"");gsub(/\(.*\)/,"");gsub(/^[ \t]+|[ \t]+$/,"");printf "%2d. %s\n",idx++,$0}
/^[ \t]+/{gsub(/<.*>/,"");gsub(/\(.*\)/,"");gsub(/^[ \t]+|[ \t]+$/,"");printf "├─%s\n",$0}
' >> "$REPORT"; then
echo "⚠️ cflow 未安装或解析失败，已跳过调用树"
fi

# 6. 生成符号列表部分（中文归类，一行三列）
echo "🔄生成符号列表..."
ctags --output-format=json \
--kinds-c=+c+d+e+f+g+l+m+n+p+s+t+u+v+x \
--kinds-c++=+c+d+e+f+g+l+m+n+p+s+t+u+v+x \
--fields=+iafkmnsSt \
--extras=+F+q+r \
"$SRC" 2>/dev/null \
| jq -r 'select(.name|startswith("__anon")|not)|(.kind+"\t"+.name)' \
| awk -v OFS="" '
BEGIN{
map["f"]="函数"; map["p"]="原型"; map["d"]="宏定义"
map["s"]="结构体"; map["u"]="联合体"; map["e"]="枚举"
map["g"]="枚举值"; map["t"]="类型"; map["v"]="变量"
map["l"]="局部"; map["h"]="头文件"; map["x"]="外部变量"
}
{
k=(($1 in map)?map[$1]:$1);
list[k]=list[k] (list[k]?" ":"") $2
}
END{
print "\n---\n## 二、符号列表"
idx=0
for(k in list){
n=split(list[k],arr," ")
printf "\n**%s**：\n",k
for(i=1;i<=n;i++){
idx++
printf " %2d. %-22s",idx,arr[i]
if(i%3==0) printf "\n"
}
if(n%3) printf "\n"
}
}' >> "$REPORT"

# 7. 输出完成信息
echo "✅ 中文分析报告已生成：$REPORT"

#!/data/data/com.termux/files/usr/bin/bash
# 1. 脚本名称：scan.sh —— Termux 交互式目录层级扫描器
# 2. 依赖：tree（如未安装请先pkg install tree）
# 3. 设置Bash 严格模式：遇到错误立即退出、使用未定义变量时报错、管道出错也整 体失败
set -euo pipefail

# 4. 定义颜色函数，让输出更醒目
red() { printf '\033[31m%b\033[0m\n' "$1"; }
green() { printf '\033[32m%b\033[0m\n' "$1"; }
yellow() { printf '\033[33m%b\033[0m\n' "$1"; }

# 5. 检查tree 是否已安装
if ! command -v tree >/dev/null 2>&1; then
red "错误：未检测到tree 命令，请先执行pkg install tree"
exit 1
fi

# 6. 主逻辑封装成函数，方便后续复用
main() {
# 7. 友好提示：让用户输入待扫描目录
echo
yellow "=== Termux 目录扫描器==="
read -rp "请输入想要扫描的目录（绝对或相对路径）: " target_dir

# 8. 把用户输入的路径转换成绝对路径，并检查是否存在
target_dir=$(realpath -m "$target_dir" 2>/dev/null || readlink -f "$target_dir" 2>/dev/null || echo "$target_dir")
if [[ ! -d $target_dir ]]; then
red "错误：目录不存在→$target_dir"
exit 1
fi

# 9. 让用户选择扫描层级（空回车默认全部层级）
read -rp "请选择扫描层级（数字，直接回车=全部层级）: " depth
if [[ -z $depth ]]; then
depth_flag=""
elif [[ $depth =~ ^[0-9]+$ ]]; then
depth_flag="-L $depth"
else
red "错误：层级必须是数字"
exit 1
fi

# 10. 执行tree 并高亮输出
green "扫描结果："
echo "----------------------------------------"
tree -C $depth_flag "$target_dir"
echo "----------------------------------------"
}

# 11. 调用主函数
main
